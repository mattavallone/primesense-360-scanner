cmake_minimum_required(VERSION 3.10)
project(3dscan_to_stl)

set(CMAKE_CXX_STANDARD 11)
set(CMAKE_CXX_STANDARD_REQUIRED True)

set(OpenCV_DIR "C:/opencv/build" CACHE PATH "OpenCV CMake package directory")
set(OpenNI_DIR "C:/Program Files/OpenNI2" CACHE PATH "OpenNI2 installation")

find_package(OpenCV REQUIRED)
find_library(OpenNI2_LIBRARY OpenNI2 HINTS "${OpenNI_DIR}/Lib")
if(NOT OpenNI2_LIBRARY)
    message(FATAL_ERROR
        "OpenNI2 library not found. Set OpenNI_DIR to its installation.")
endif()

add_executable(3dscan_to_stl main.cpp scan_mesh.cpp)
target_include_directories(3dscan_to_stl PRIVATE
    "${OpenNI_DIR}/Include"
    ${OpenCV_INCLUDE_DIRS}
)
target_link_libraries(3dscan_to_stl PRIVATE ${OpenCV_LIBS} ${OpenNI2_LIBRARY})
if(WIN32)
    target_link_libraries(3dscan_to_stl PRIVATE Comdlg32)
endif()
set_target_properties(3dscan_to_stl PROPERTIES
    RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin"
)

set(PRIMESENSE_RUNTIME_DIR
    "${CMAKE_SOURCE_DIR}/components/primesense-openni2-winusb")
if(WIN32)
    find_program(POWERSHELL_EXECUTABLE NAMES powershell.exe powershell)
    if(NOT POWERSHELL_EXECUTABLE)
        message(FATAL_ERROR
            "PowerShell is required to bundle the OpenNI2 runtime.")
    endif()

    add_custom_command(TARGET 3dscan_to_stl POST_BUILD
        COMMAND "${POWERSHELL_EXECUTABLE}" -NoProfile -ExecutionPolicy Bypass
            -File "${PRIMESENSE_RUNTIME_DIR}/Prepare-OpenNI2-Runtime.ps1"
            -RedistDir "${OpenNI_DIR}/Redist"
            -OutputDir "$<TARGET_FILE_DIR:3dscan_to_stl>"
        VERBATIM
    )

    add_custom_target(build_winusb_openni
        COMMAND "${POWERSHELL_EXECUTABLE}" -NoProfile -ExecutionPolicy Bypass
            -File "${PRIMESENSE_RUNTIME_DIR}/Build-OpenNI2-WinUSB.ps1"
            -BuildDir "${CMAKE_BINARY_DIR}"
            -OutputDriverDir "$<TARGET_FILE_DIR:3dscan_to_stl>/OpenNI2/Drivers"
        VERBATIM
    )
    add_dependencies(build_winusb_openni 3dscan_to_stl)
endif()

option(BUILD_TESTING "Build scanner mesh tests" OFF)
if(BUILD_TESTING)
    enable_testing()
    add_executable(scan_mesh_tests tests/scan_mesh_tests.cpp scan_mesh.cpp)
    target_include_directories(scan_mesh_tests PRIVATE "${CMAKE_SOURCE_DIR}")
    add_test(NAME scan_mesh_tests COMMAND scan_mesh_tests)
endif()
